    // Coach/admin edits overlay — everyone signed in can READ (so applied
    // edits show for players too); only staff (listed in /staff) can WRITE.
    match /edits/{docId} {
      allow read:  if request.auth != null;
      allow write: if request.auth != null
                   && exists(/databases/$(database)/documents/staff/$(request.auth.token.email));
    }

    // Staff allowlist. Readable by signed-in users; only editable from the
    // Firebase console (never from the app) so it can't be tampered with.
    match /staff/{email} {
      allow read:  if request.auth != null;
      allow write: if false;
    }
